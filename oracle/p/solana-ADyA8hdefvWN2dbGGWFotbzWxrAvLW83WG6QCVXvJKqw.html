Reproduce the independent read yourself (4 public RPC calls, no key)

This reads the program that holds the account, then its ProgramData, then the Squads v4 committee that is the upgrade authority. Each call is a plain JSON-RPC request to a public Solana RPC; the last step, when there is one, is a short Python 3 program with no library.

# 1. The account is owned by a program (value.owner): pAMMBay6oceH9fJKBRHGP5D4bD4sWpmSwMn52FMfXEA
curl -s -X POST https://api.mainnet-beta.solana.com -H 'content-type: application/json' -d '{"jsonrpc":"2.0","id":1,"method":"getAccountInfo","params":["ADyA8hdefvWN2dbGGWFotbzWxrAvLW83WG6QCVXvJKqw",{"encoding":"jsonParsed"}]}'

# 2. The program: its ProgramData account is the one that holds the upgrade authority (value.data.parsed.info.programData)
curl -s -X POST https://api.mainnet-beta.solana.com -H 'content-type: application/json' -d '{"jsonrpc":"2.0","id":1,"method":"getAccountInfo","params":["pAMMBay6oceH9fJKBRHGP5D4bD4sWpmSwMn52FMfXEA",{"encoding":"jsonParsed"}]}'

# 3. The ProgramData: the upgrade authority (value.data.parsed.info.authority)

import hashlib as H
A = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz"; P = 2**255 - 19; D = -121665 * pow(121666, P - 2, P) % P
dec = lambda s: int(sum(A.index(c) * 58**i for i, c in enumerate(reversed(s)))).to_bytes(32, "big")
def enc(b):
    n = int.from_bytes(b, "big"); s = ""
    while n: n, r = divmod(n, 58); s = A[r] + s
    return "1" * (len(b) - len(b.lstrip(b"\0"))) + s
def on_curve(b):
    y = int.from_bytes(b, "little") & ((1 << 255) - 1)
    if y >= P: return False
    x2 = (y * y - 1) * pow(D * y * y + 1, P - 2, P) % P
    return x2 == 0 or pow(x2, (P - 1) // 2, P) == 1
for bump in range(255, -1, -1):
