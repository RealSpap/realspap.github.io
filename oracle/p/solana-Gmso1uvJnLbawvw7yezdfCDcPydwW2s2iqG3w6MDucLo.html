Reproduce the independent read yourself (3 public RPC calls, no key)

import hashlib as H
A = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz"; P = 2**255 - 19; D = -121665 * pow(121666, P - 2, P) % P
dec = lambda s: int(sum(A.index(c) * 58**i for i, c in enumerate(reversed(s)))).to_bytes(32, "big")
def enc(b):
    n = int.from_bytes(b, "big"); s = ""
    while n: n, r = divmod(n, 58); s = A[r] + s
    return "1" * (len(b) - len(b.lstrip(b"\0"))) + s
def on_curve(b):
    y = int.from_bytes(b, "little") & ((1 << 255) - 1)
    if y >= P: return False
    x2 = (y * y - 1) * pow(D * y * y + 1, P - 2, P) % P
    return x2 == 0 or pow(x2, (P - 1) // 2, P) == 1
for bump in range(255, -1, -1):
